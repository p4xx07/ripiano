export async function open({ mobile = false, locale = "it-IT", storageState } = {}) {
  const browser = await chromium.launch({ channel: "chrome", headless: true });
  const ctx = await browser.newContext(mobile
    ? { viewport: { width: 390, height: 844 }, isMobile: true, hasTouch: true, locale, storageState, baseURL: ORIGIN }
    : { viewport: { width: 1440, height: 900 }, locale, storageState, baseURL: ORIGIN });
  const page = await ctx.newPage();
  const logs = [];
  page.on("console", m => { if (m.type() === "error" || m.type() === "warning") logs.push(`[${m.type()}] ${m.text()}`); });
  page.on("pageerror", e => logs.push(`[pageerror] ${e.message}`));
  page.on("response", r => { if (r.status() >= 400) logs.push(`[http ${r.status()}] ${r.request().method()} ${r.url()}`); });
  return { browser, ctx, page, logs };
}

export async function audit(page) {
  return page.evaluate(() => {
    const problems = [];
    const visible = (el) => el.getClientRects().length > 0 && getComputedStyle(el).visibility !== "hidden";
    const short = (el) => el.outerHTML.slice(0, 110);
    for (const el of document.querySelectorAll("button, a[href], [role=button]")) {
      if (!visible(el)) continue;
      const name = (el.getAttribute("aria-label") || el.textContent || el.getAttribute("title") || "").trim();
      if (!name) problems.push(`no accessible name: ${short(el)}`);
      const r = el.getBoundingClientRect();
      if (innerWidth < 500 && (r.width < 40 || r.height < 40) && r.width > 0) problems.push(`small touch target ${Math.round(r.width)}x${Math.round(r.height)}: ${(el.getAttribute("aria-label") || el.textContent || "").trim().slice(0, 40)}`);
    }
    for (const el of document.querySelectorAll("input:not([type=hidden]), select, textarea")) {
      if (!visible(el)) continue;
      const labelledBy = (el.getAttribute("aria-labelledby") || "").split(" ").map((id) => document.getElementById(id)?.textContent ?? "").join("");
      const forLabel = el.id ? document.querySelector(`label[for="${CSS.escape(el.id)}"]`)?.textContent ?? "" : "";
      const label = el.getAttribute("aria-label") || labelledBy || forLabel || el.closest("label")?.textContent;
      if (!label?.trim()) problems.push(`form field without real label: ${short(el)}`);
    }
    const ids = new Map();
    for (const el of document.querySelectorAll("[id]")) ids.set(el.id, (ids.get(el.id) ?? 0) + 1);
    for (const [id, count] of ids) if (count > 1) problems.push(`duplicate id "${id}" x${count}`);
    for (const img of document.querySelectorAll("img")) if (visible(img) && !img.hasAttribute("alt")) problems.push(`image without alt: ${short(img)}`);
    let last = 0;
    for (const heading of document.querySelectorAll("h1, h2, h3, h4, h5, h6")) {
      if (!visible(heading)) continue;
      const level = Number(heading.tagName[1]);
      if (last && level > last + 1) problems.push(`heading jumps h${last}->h${level}: ${heading.textContent?.trim().slice(0, 40)}`);
      last = level;
    }
    if (document.querySelectorAll("h1").length !== 1) problems.push(`h1 count ${document.querySelectorAll("h1").length}`);
    if (!document.querySelector("main")) problems.push("no <main>");
    problems.push(`lang=${document.documentElement.lang} title=${document.title}`);
    if (document.documentElement.scrollWidth > innerWidth + 1) problems.push(`horizontal overflow ${document.documentElement.scrollWidth} > ${innerWidth}`);
    return problems;
  });
}

async function mcp(token, calls) {

  if (r.status !== 200) return [{ initStatus: r.status, body: (await r.text()).slice(0, 200), www: r.headers.get("www-authenticate") }];
  const sid = r.headers.get("mcp-session-id"); await parse(r);
  if (sid) H["Mcp-Session-Id"] = sid;
