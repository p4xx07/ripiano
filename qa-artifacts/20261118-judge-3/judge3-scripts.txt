export function password(key = "JUDGE3_PASSWORD") {
  const line = fs.readFileSync(`${SCR}/judges.env`, "utf8").split("\n").find((l) => l.startsWith(key + "="));
  return line.slice(key.length + 1).trim().replace(/^['"]|['"]$/g, "");
}
export function mask(s) { return String(s).replace(/(rip_|rpa_|ripa_)[A-Za-z0-9_\-]+/g, "$1****").replace(/[A-Za-z0-9_\-]{40,}/g, "****"); }
// API context with a session cookie and CSRF header helper
export async function apiSession(username = "judge3", pw = password()) {
  const ctx = await pwRequest.newContext({ baseURL: BASE, extraHTTPHeaders: { Origin: BASE, Referer: BASE + "/" } });
  await ctx.get("/api/session");
  let csrf = (await ctx.storageState()).cookies.find((c) => c.name === "csrftoken")?.value;
  const r = await ctx.post("/api/session", { data: { username, password: pw }, headers: csrf ? { "X-CSRFToken": csrf } : {} });
  if (!r.ok()) throw new Error("login failed " + r.status() + " " + (await r.text()));
  csrf = (await ctx.storageState()).cookies.find((c) => c.name === "csrftoken")?.value;
  const h = () => ({ "X-CSRFToken": csrf });
  return {
    ctx,
    get: (u) => ctx.get(u),
    post: (u, data, extra = {}) => ctx.post(u, { data, headers: { ...h(), ...(extra.headers || {}) }, ...extra }),
    patch: (u, data) => ctx.patch(u, { data, headers: h() }),
    del: (u) => ctx.delete(u, { headers: h() }),
    multipart: (u, multipart) => ctx.post(u, { multipart, headers: h() }),
  };
}

}
