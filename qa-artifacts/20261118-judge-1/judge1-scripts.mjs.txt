export async function launch(mobile=false, opts={}) {
  const browser = await chromium.launch({ channel: "chrome", headless: true });
  const ctx = await browser.newContext(mobile ? { viewport:{width:390,height:844}, isMobile:true, hasTouch:true, deviceScaleFactor:3, userAgent: devices["iPhone 13"].userAgent, ...opts } : { viewport:{width:1440,height:900}, ...opts });
  const page = await ctx.newPage();
  page.on("pageerror", e => console.log("PAGEERROR", e.message));
  page.on("console", m => { if (m.type()==="error") console.log("CONSOLE", m.text()); });
  page.on("response", r => { if (r.status()>=500) console.log("HTTP", r.status(), r.url()); });
  return { browser, ctx, page };
}

// ===== mcp.mjs =====
import { readFileSync } from "node:fs";
const tokens = JSON.parse(readFileSync(new URL("./agent-tokens.json", import.meta.url)));
export async function rpc(agent, method, params, id=1) {
  const r = await fetch("http://localhost:18080/mcp", { method:"POST", headers:{ "Content-Type":"application/json", "Accept":"application/json, text/event-stream", "Authorization":"Bearer "+tokens[agent], "MCP-Protocol-Version":"2025-06-18" }, body: JSON.stringify({ jsonrpc:"2.0", id, method, params }) });
  const t = await r.text();
  const line = t.split("\n").find(l=>l.startsWith("data:"));
  let body; try { body = JSON.parse(line ? line.slice(5) : t); } catch { body = t.slice(0,300); }
  return { status: r.status, body };
}
